=== LLAMADA POWERSHELL (Host Windows - Invoke-RestMethod) ===
PS C:\Users\Sebas> $chatBody = @{
    model = "asistente-ciberseguridad"
    messages = @(
        @{ role = "user"; content = "¿Qué es un firewall? Responde de forma técnica y concisa en 3 viñetas." }
    )
    stream = $false
} | ConvertTo-Json
PS C:\Users\Sebas> $chatResp = Invoke-RestMethod -Uri "http://127.0.0.1:11434/api/chat" -Method Post -ContentType "application/json" -Body $chatBody
